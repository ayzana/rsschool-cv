            function invertCase(str) {
              let result = "";
              for (let i = 0; i < str.length; i += 1) {
                let letter = str.charAt(i);
                if (letter === letter.toUpperCase()) {
                  letter = letter.toLowerCase();
                } else {
                  letter = letter.toUpperCase();
                }
                result += letter;
              }
              return result;
            }
